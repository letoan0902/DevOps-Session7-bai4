# ===== Phần 4: Tạo file cấu hình spring-proxy.conf (09:14) =====

devops@halo:~$ sudo tee /etc/nginx/sites-available/spring-proxy.conf > /dev/null << 'EOF'
> # /etc/nginx/sites-available/spring-proxy.conf
> # Server block làm Reverse Proxy cho ứng dụng Spring Boot (Session 07 - Bài 4).
> #   - Đường dẫn "/"      : Nginx phục vụ trực tiếp trang tĩnh trong /var/www/html.
> #   - Đường dẫn "/api/"  : Nginx chuyển tiếp (proxy_pass) đến Spring Boot ở 127.0.0.1:8082.
> # Site này thay thế site my-app trên cổng 80, nên phải gỡ symlink my-app.conf
> # khỏi sites-enabled trước khi kích hoạt (xem README).
>
> server {
>     # Lắng nghe cổng 80 cho IPv4 và IPv6.
>     # default_server: nhận mọi yêu cầu không khớp server_name nào,
>     # ví dụ truy cập trực tiếp bằng địa chỉ IP của máy chủ.
>     listen 80 default_server;
>     listen [::]:80 default_server;
>
>     # Tên miền "_" là giá trị giữ chỗ, không khớp tên miền thật nào.
>     server_name _;
>
>     # Thư mục chứa trang tĩnh và file chỉ mục mặc định.
>     root /var/www/html;
>     index index.html;
>
>     # Log riêng cho site này để dễ lọc khi chẩn đoán sự cố.
>     access_log /var/log/nginx/spring-proxy.access.log;
>     error_log  /var/log/nginx/spring-proxy.error.log;
>
>     # Khối 1 - Lưu lượng tĩnh.
>     # Chỉ trả file có thật trong root, nếu không có thì trả 404.
>     location / {
>         try_files $uri $uri/ =404;
>     }
>
>     # Khối 2 - Lưu lượng động (Reverse Proxy).
>     # Mọi yêu cầu bắt đầu bằng /api/ được chuyển đến backend Spring Boot.
>     location /api/ {
>         # Dấu "/" ở cuối URL đích có chủ ý: Nginx thay phần "/api/" bằng "/"
>         # trước khi gửi đi, nên /api/health thành /health ở backend.
>         # Nếu bỏ dấu "/" cuối thì backend nhận nguyên /api/health và trả 404.
>         proxy_pass http://127.0.0.1:8082/;
>
>         # Dùng HTTP/1.1 để nói chuyện với backend (mặc định Nginx dùng HTTP/1.0),
>         # cho phép giữ kết nối keep-alive và chunked transfer.
>         proxy_http_version 1.1;
>
>         # Chuyển tiếp thông tin của client gốc cho backend.
>         # Nếu thiếu, backend chỉ thấy Host là 127.0.0.1:8082 và IP client là 127.0.0.1.
>         proxy_set_header Host              $host;
>         proxy_set_header X-Real-IP         $remote_addr;
>         proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
>         proxy_set_header X-Forwarded-Proto $scheme;
>
>         # Giới hạn thời gian: kết nối tới backend tối đa 5 giây,
>         # chờ phản hồi tối đa 60 giây trước khi trả 504 Gateway Timeout.
>         proxy_connect_timeout 5s;
>         proxy_read_timeout    60s;
>     }
> }
> EOF
